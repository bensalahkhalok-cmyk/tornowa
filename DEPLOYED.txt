🔄 API ENDPOINTS
═════════════════════════════════════════════════════════════════════

Authentication:
  POST   /auth/register          - Sign up
  POST   /auth/login             - Sign in
  GET    /auth/profile           - Get profile
  PUT    /auth/profile           - Update profile

Leagues:
  POST   /leagues                - Create league
  GET    /leagues                - Get all leagues
  GET    /leagues/:id            - Get league details
  PUT    /leagues/:id            - Update league
  DELETE /leagues/:id            - Delete league
  POST   /leagues/:id/generate-invite  - Create invite
  POST   /leagues/verify-invite  - Join league

Teams:
  POST   /teams                  - Create team
  GET    /teams                  - Get all teams
  GET    /teams/:id              - Get team details
  POST   /teams/:id/add-player   - Add player
  POST   /teams/:id/accept-membership - Confirm

Matches:
  POST   /matches                - Create match
  GET    /matches                - Get all matches
  GET    /matches/:id            - Get match details
  PUT    /matches/:id            - Update match/score
  POST   /matches/:id/events     - Record event

Standings:
  GET    /standings              - Get league standings
